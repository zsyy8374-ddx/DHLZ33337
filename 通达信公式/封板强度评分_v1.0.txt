{===============================================
  封板强度评分 v1.0 (Dengxian 5-3 委托)
  作者: dengxian (基于 5-3 教的 GPJYVALUE 函数)
  
  目的: 综合评估当日涨停股的"封板硬度", 用于:
    1. 次日开板风险预警 (评分低 = 次日易炸板)
    2. 板块内龙头识别 (评分高的是真龙头)
    3. 接力候选筛选 (评分≥80 才考虑接力)
  
  维度:
    - 首封时间 (越早越强)
    - 炸板次数 (0次最佳)
    - 盘中最大封单 vs 流通盘 (主力实力)
    - 封板持续度 = 收盘封单/最大封单 (是否被撬过)
    - 板上成交额占全天比 (低=真封, 高=博弈)
===============================================}

{----- 1. 基础涨停判断 (含 ST 5%, 双创 20%) -----}
涨幅:=IF(NAMEINCLUDE('ST'),0.05,IF(CODELIKE('30') OR CODELIKE('68'), 0.2, 0.1));
涨停价:=ROUND2(REF(C,1)*(1+涨幅), 2);
ZT:=C>=涨停价 AND C=H;

{----- 2. 调用 GPJYVALUE 取盘中数据 -----}
首封时间:=GPJYVALUE(24,1,2);    {格式: HHMMSS, 如 93000 = 9:30:00}
最大封单万:=GPJYVALUE(24,2,2);  {盘中峰值封单, 单位万元}
板上成交万:=GPJYVALUE(14,1,2);  {以涨停价成交的金额, 万元}
开板次数:=GPJYVALUE(14,2,2);    {当日开板次数}
收盘封单万:=GPJYVALUE(15,2,2);  {收盘封单额, 万元}

{----- 3. 衍生指标 -----}
{首封时间转分钟数, 越早越强}
首封小时:=INTPART(首封时间/10000);
首封分钟:=INTPART((首封时间-首封小时*10000)/100);
首封总分钟:=首封小时*60+首封分钟;

{封板持续度: 收盘/最大, 越接近1越好 (一直封死)}
封板持续度:=IF(最大封单万>0, 收盘封单万/最大封单万, 0);

{板上成交占比: 板上成交/全天成交, 低=真封 (没多少人卖给你)}
板上占比:=IF(AMOUNT>0, 板上成交万*10000/AMOUNT*100, 0);

{最大封单/流通市值 比值, 越大主力越凶}
流通市值万:=CAPITAL*C/10000;
最大封单占比:=IF(流通市值万>0, 最大封单万/流通市值万*100, 0);

{----- 4. 五维度评分 (各 20 分, 总 100) -----}

{① 首封时间分: 9:30=20, 10:00=15, 10:30=10, 11:30=5, 14:30+=0}
S首封:=IF(首封总分钟<=570, 20,         {9:30 之前 (集合竞价后秒板)}
      IF(首封总分钟<=600, 18,         {10:00}
      IF(首封总分钟<=630, 15,         {10:30}
      IF(首封总分钟<=690, 10,         {11:30}
      IF(首封总分钟<=810, 5,          {13:30}
      IF(首封总分钟<=870, 2, 0))))));  {14:30 后给 2-0 分}

{② 炸板次数: 0=20, 1=12, 2=6, 3+=0}
S炸板:=IF(开板次数=0, 20,
      IF(开板次数=1, 12,
      IF(开板次数=2, 6,
      IF(开板次数=3, 2, 0))));

{③ 最大封单占流通市值比 (主力实力, 单位 %)}
{≥3%=20, 1-3%=15, 0.5-1%=10, 0.2-0.5%=5, <0.2%=0}
S封单:=IF(最大封单占比>=3, 20,
      IF(最大封单占比>=1, 15,
      IF(最大封单占比>=0.5, 10,
      IF(最大封单占比>=0.2, 5, 0))));

{④ 封板持续度: ≥0.9=20, 0.7-0.9=15, 0.5-0.7=10, 0.3-0.5=5, <0.3=0}
S持续:=IF(封板持续度>=0.9, 20,
      IF(封板持续度>=0.7, 15,
      IF(封板持续度>=0.5, 10,
      IF(封板持续度>=0.3, 5, 0))));

{⑤ 板上成交占比反向打分 (低=好): <5%=20, 5-15%=15, 15-30%=10, 30-50%=5, >50%=0}
S板上:=IF(板上占比<5, 20,
      IF(板上占比<15, 15,
      IF(板上占比<30, 10,
      IF(板上占比<50, 5, 0))));

{----- 5. 总分输出 -----}
封板总分: ZT * (S首封 + S炸板 + S封单 + S持续 + S板上);

{----- 6. 选股条件 -----}
{S 档 (≥85): 极强封板, 次日不开板大概率, 真龙头}
{A 档 (70-84): 强封板, 接力可考虑}
{B 档 (50-69): 中等封板, 需配合板块}
{C 档 (<50): 弱封板, 次日炸板风险, 避雷}

S档:= ZT AND 封板总分>=85;
A档:= ZT AND 封板总分>=70 AND 封板总分<85;
B档:= ZT AND 封板总分>=50 AND 封板总分<70;
C档:= ZT AND 封板总分<50;

{----- 7. 副图显示 -----}
DRAWNUMBER(ZT, H*1.02, 封板总分),COLORYELLOW;
DRAWICON(S档, L*0.98, 1);   {S档红色三角}
DRAWICON(A档, L*0.98, 2);   {A档橙色}
DRAWICON(B档, L*0.98, 3);   {B档黄色}

{----- 8. 选股用主条件 -----}
{选股入口: 输出 1 表示当日入选}
SELECT: 封板总分;
