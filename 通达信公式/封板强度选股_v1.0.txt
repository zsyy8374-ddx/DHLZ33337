{===============================================
  封板强度选股 v1.0 - 简化选股版
  
  使用步骤 (在 XTXBDDX 通达信里):
  1. Ctrl+F (公式管理器) → 新建 → 条件选股公式
  2. 名称: 封板强度评分
  3. 粘贴本公式
  4. 测试公式 → 保存
  5. Ctrl+T (条件选股) → 选本公式 → 选最近交易日 → 执行
  6. 结果右键 "导出数据为 CSV" → 发我 (邮件或微信)
  
  输出 5 个字段:
  - 总分 (0-100)
  - 首封时间 (分钟, 9:30=570)
  - 炸板次数
  - 最大封单(万)
  - 持续度 (×100, 100=完全持续)
===============================================}

涨幅:=IF(NAMEINCLUDE('ST'),0.05,IF(CODELIKE('30') OR CODELIKE('68'), 0.2, 0.1));
涨停价:=ROUND2(REF(C,1)*(1+涨幅), 2);
ZT:=C>=涨停价 AND C=H;

{GPJYVALUE 调用}
首封时间:=GPJYVALUE(24,1,2);
最大封单万:=GPJYVALUE(24,2,2);
板上成交万:=GPJYVALUE(14,1,2);
开板次数:=GPJYVALUE(14,2,2);
收盘封单万:=GPJYVALUE(15,2,2);

{衍生指标}
首封小时:=INTPART(首封时间/10000);
首封分:=INTPART((首封时间-首封小时*10000)/100);
首封总分钟:=首封小时*60+首封分;

封板持续度:=IF(最大封单万>0, 收盘封单万/最大封单万, 0);
板上占比:=IF(AMOUNT>0, 板上成交万*10000/AMOUNT*100, 0);
流通市值万:=CAPITAL*C/10000;
最大封单占比:=IF(流通市值万>0, 最大封单万/流通市值万*100, 0);

{五维评分}
S首封:=IF(首封总分钟<=570, 20, IF(首封总分钟<=600, 18, IF(首封总分钟<=630, 15, IF(首封总分钟<=690, 10, IF(首封总分钟<=810, 5, IF(首封总分钟<=870, 2, 0))))));
S炸板:=IF(开板次数=0, 20, IF(开板次数=1, 12, IF(开板次数=2, 6, IF(开板次数=3, 2, 0))));
S封单:=IF(最大封单占比>=3, 20, IF(最大封单占比>=1, 15, IF(最大封单占比>=0.5, 10, IF(最大封单占比>=0.2, 5, 0))));
S持续:=IF(封板持续度>=0.9, 20, IF(封板持续度>=0.7, 15, IF(封板持续度>=0.5, 10, IF(封板持续度>=0.3, 5, 0))));
S板上:=IF(板上占比<5, 20, IF(板上占比<15, 15, IF(板上占比<30, 10, IF(板上占比<50, 5, 0))));

总分: ZT * (S首封 + S炸板 + S封单 + S持续 + S板上);

{选股: 只选当日涨停且总分≥50}
ZT AND 总分>=50;

{副图字段, 导出 CSV 时一起出来}
首封分钟: 首封总分钟;
炸板: 开板次数;
最大封单: 最大封单万;
持续度: 封板持续度*100;
